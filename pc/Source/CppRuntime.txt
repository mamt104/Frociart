class BigNat {

inline bool runWithStack(const std::function<bool()>& action){
 struct Work{std::function<bool()> action;std::atomic<bool> result{false};explicit Work(std::function<bool()> f):action(std::move(f)){} };Work work(action);
#ifdef _WIN32
 struct Entry{static unsigned __stdcall start(void* pointer){auto& work=*static_cast<Work*>(pointer);work.result.store(work.action(),std::memory_order_release);return 0;}};
 auto handle=reinterpret_cast<HANDLE>(_beginthreadex(nullptr,16*1024*1024,Entry::start,&work,STACK_SIZE_PARAM_IS_A_RESERVATION,nullptr));if(!handle)throw Problem("Non riesco a preparare lo spazio per le chiamate di funzione.");if(WaitForSingleObject(handle,INFINITE)!=WAIT_OBJECT_0)std::terminate();CloseHandle(handle);
#else
 struct Entry{static void* start(void* pointer){auto& work=*static_cast<Work*>(pointer);work.result.store(work.action(),std::memory_order_release);return nullptr;}};
 pthread_attr_t attributes;if(pthread_attr_init(&attributes)!=0)throw Problem("Non riesco a preparare l’esecuzione.");if(pthread_attr_setstacksize(&attributes,16*1024*1024)!=0){pthread_attr_destroy(&attributes);throw Problem("Non riesco a preparare lo spazio per le chiamate di funzione.");}pthread_t thread;int result=pthread_create(&thread,&attributes,Entry::start,&work);pthread_attr_destroy(&attributes);if(result!=0)throw Problem("Non riesco ad avviare l’esecuzione.");if(pthread_join(thread,nullptr)!=0)std::terminate();
#endif
 return work.result.load(std::memory_order_acquire);
}

class Turtle {
 static std::string number(double value){if(value==0)return "0";std::ostringstream out;out.imbue(std::locale::classic());out<<std::setprecision(17)<<value;return out.str();}
public:
 struct Segment{double x1,y1,x2,y2;std::string color;};
 double x=0,y=0,heading=90;std::string color="#000000";bool used=false;std::vector<Segment> segments;
 static std::string parseColor(const Value& value){auto text=asciiLower(trim(value.asText()));static const char* names[][3]={{"nero","black","#000000"},{"bianco","white","#FFFFFF"},{"grigio","gray","#808080"},{"rosso","red","#FF0000"},{"arancione","orange","#FF8C00"},{"giallo","yellow","#FFFF00"},{"verde","green","#008000"},{"lime","lime","#00FF00"},{"turchese","teal","#008080"},{"ciano","cyan","#00FFFF"},{"blu","blue","#0000FF"},{"azzurro","skyblue","#00BFFF"},{"viola","purple","#800080"},{"magenta","magenta","#FF00FF"},{"rosa","pink","#FF69B4"},{"marrone","brown","#A52A2A"}};for(const auto& row:names)if(text==fromUtf8(row[0])||text==fromUtf8(row[1]))return row[2];if(text==u"grey")return "#808080";bool valid=(text.size()==4||text.size()==7)&&text[0]==u'#';for(std::size_t i=1;i<text.size();++i)valid&=(text[i]>=u'0'&&text[i]<=u'9')||(text[i]>=u'a'&&text[i]<=u'f');if(!valid)throw Problem("Colore non riconosciuto. Usa un nome o #RRGGBB.");if(text.size()==4)text=Text{u'#',text[1],text[1],text[2],text[2],text[3],text[3]};for(auto& c:text)if(c>=u'a'&&c<=u'f')c-=32;return utf8(text);}
 void apply(int kind,const Value& value,bool flag){if(kind==12||kind==13){double n=value.asNumber();if(kind==12){double rad=heading*std::acos(-1.0)/180,cos=heading==90||heading==270?0:heading==0?1:heading==180?-1:std::cos(rad),sin=heading==0||heading==180?0:heading==90?1:heading==270?-1:std::sin(rad),nx=x+n*cos,ny=y+n*sin;if(!std::isfinite(nx)||!std::isfinite(ny)||std::abs(nx)>1000000||std::abs(ny)>1000000)throw Problem("Il disegno supera le coordinate tra -1000000 e 1000000.");bool line=flag&&(nx!=x||ny!=y);if(line&&segments.size()>=10000)throw Problem("Il disegno supera 10000 tratti.");if(line)segments.push_back({x,y,nx,ny,color});x=nx;y=ny;}else{double angle=std::fmod(n,360.0);heading=std::fmod(std::fmod(heading+(flag?angle:-angle),360.0)+360,360.0);}}
  else if(kind==14||kind==15){x=y=0;heading=90;if(kind==15)segments.clear();}else if(kind==16)color=parseColor(value);else throw Problem("Comando tartaruga non riconosciuto.");used=true;
 }
 std::string svg()const{double minX=std::min(0.0,x),maxX=std::max(0.0,x),minY=std::min(0.0,y),maxY=std::max(0.0,y);for(const auto& s:segments){minX=std::min(minX,std::min(s.x1,s.x2));maxX=std::max(maxX,std::max(s.x1,s.x2));minY=std::min(minY,std::min(s.y1,s.y2));maxY=std::max(maxY,std::max(s.y1,s.y2));}double margin=std::max(10.0,std::max(maxX-minX,maxY-minY)*.05),width=std::max(1.0,maxX-minX)+2*margin,height=std::max(1.0,maxY-minY)+2*margin;std::string out="<svg xmlns=\"http://www.w3.org/2000/svg\" viewBox=\""+number(minX-margin)+" "+number(-maxY-margin)+" "+number(width)+" "+number(height)+"\"><title>Disegno tartaruga — Frociart</title><rect x=\""+number(minX-margin)+"\" y=\""+number(-maxY-margin)+"\" width=\""+number(width)+"\" height=\""+number(height)+"\" fill=\"white\"/>";for(const auto& s:segments)out+="<line x1=\""+number(s.x1)+"\" y1=\""+number(-s.y1)+"\" x2=\""+number(s.x2)+"\" y2=\""+number(-s.y2)+"\" stroke=\""+s.color+"\" stroke-width=\"2\" stroke-linecap=\"round\" vector-effect=\"non-scaling-stroke\"/>";return out+"</svg>";}
};
